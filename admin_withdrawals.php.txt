    <script>
        function renderRequests() {
            let requests = JSON.parse(localStorage.getItem('lottery_admin_withdrawals') || '[]');
            let container = document.getElementById("adminListContainer");

            if (requests.length === 0) {
                container.innerHTML = `<p style="font-size:12px; color:#888; text-align:center; padding:15px;">No withdrawal requests found.</p>`;
                return;
            }

            let html = "";
            requests.forEach((req, index) => {
                let badgeColor = req.status === 'Paid' ? '#00c853' : '#ff4d4f';
                html += `
                    <div class="admin-req-card">
                        <div>
                            <b>ID: ${req.id}</b><br>
                            Amount: <b style="color:green; font-size:14px;">₹${req.amount}</b><br>
                            Bank: ${req.bank}<br>
                            Acc No: <b style="color:#000; background:#eee; padding:2px 4px;">${req.acc}</b><br>
                            IFSC: ${req.ifsc} | Name: ${req.holder}
                        </div>
                        <div style="text-align:right;">
                            <span style="color:${badgeColor}; font-weight:bold; display:block; margin-bottom:8px;">${req.status}</span>
                            ${req.status === 'Pending' ? `<button onclick="approvePayment(${index})" class="btn-pay">Approve & Pay</button>` : ''}
                        </div>
                    </div>
                `;
            });
            container.innerHTML = html;
        }

        function approvePayment(index) {
            let requests = JSON.parse(localStorage.getItem('lottery_admin_withdrawals') || '[]');
            requests[index].status = "Paid";
            localStorage.setItem('lottery_admin_withdrawals', JSON.stringify(requests));
            renderRequests();
            alert("Payment Approved Successfully!");
        }

        window.onload = renderRequests;
    </script>
